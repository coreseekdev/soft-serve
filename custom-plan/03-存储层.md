# 03 - 存储层

## 设计原则

1. **保留数据库后端**: 不删除现有 `pkg/store/database/`
2. **新增文件系统后端**: 新建 `pkg/store/file/`
3. **编译时互斥**: 使用 build tags 选择后端

## Build Tags 设计

### 数据库后端

```go
// pkg/store/database/store_dbstore.go
//go:build dbstore

package database

import (
    "context"
    "github.com/charmbracelet/soft-serve/pkg/config"
    "github.com/charmbracelet/soft-serve/pkg/db"
    "github.com/charmbracelet/soft-serve/pkg/store"
)

// 确保 implements store.Store
var _ store.Store = (*DBStore)(nil)

type DBStore struct {
    db  *db.DB
    cfg *config.Config
}

func NewStore(cfg *config.Config) (store.Store, error) {
    database, err := db.Open(cfg)
    if err != nil {
        return nil, err
    }
    return &DBStore{db: database, cfg: cfg}, nil
}

// ... 实现所有 store 接口方法
```

### 文件系统后端

```go
// pkg/store/file/store_filestore.go
//go:build filestore

package file

import (
    "github.com/charmbracelet/soft-serve/pkg/config"
    "github.com/charmbracelet/soft-serve/pkg/store"
)

var _ store.Store = (*FileStore)(nil)

type FileStore struct {
    cfg        *config.Config
    users      map[string]*User
    repos      map[string]*Repo
    mu         sync.RWMutex
}

func NewStore(cfg *config.Config) (store.Store, error) {
    fs := &FileStore{
        cfg:   cfg,
        users: make(map[string]*User),
        repos: make(map[string]*Repo),
    }

    // 加载用户
    if err := fs.loadUsers(); err != nil {
        return nil, err
    }

    // 发现仓库
    if err := fs.discoverRepos(); err != nil {
        return nil, err
    }

    return fs, nil
}
```

## FileStore 实现

### 用户管理

```go
// pkg/store/file/user.go
package file

// 用户目录结构:
// {DATA_PATH}/users/
// ├── alice/
// │   └── .ssh/
// │       ├── id_ed25519.pub
// │       └── id_rsa.pub
// └── bob/
//     └── .ssh/
//         └── id_ed25519.pub

func (s *FileStore) loadUsers() error {
    usersDir := s.cfg.Paths.Users
    if usersDir == "" {
        usersDir = filepath.Join(s.cfg.DataPath, "users")
    }

    entries, err := os.ReadDir(usersDir)
    if os.IsNotExist(err) {
        return os.MkdirAll(usersDir, 0755)
    }
    if err != nil {
        return err
    }

    for _, entry := range entries {
        if !entry.IsDir() {
            continue
        }

        username := entry.Name()
        sshDir := filepath.Join(usersDir, username, ".ssh")
        keys, err := s.loadUserKeys(sshDir)
        if err != nil || len(keys) == 0 {
            continue
        }

        s.users[username] = &User{
            Username: username,
            Keys:     keys,
        }
    }

    return nil
}

func (s *FileStore) UserByUsername(ctx context.Context, username string) (*models.User, error) {
    s.mu.RLock()
    defer s.mu.RUnlock()

    user, ok := s.users[username]
    if !ok {
        return nil, store.ErrUserNotFound
    }
    return user.ToModel(), nil
}

func (s *FileStore) UserByPublicKey(ctx context.Context, key string) (*models.User, error) {
    s.mu.RLock()
    defer s.mu.RUnlock()

    for _, user := range s.users {
        for _, k := range user.Keys {
            if keysEqual(k, key) {
                return user.ToModel(), nil
            }
        }
    }
    return nil, store.ErrUserNotFound
}
```

### 仓库发现

```go
// pkg/store/file/repo.go
package file

// 自动发现 {DATA_PATH} 下的仓库
// - xxx.git/ -> bare repo
// - xxx/     -> normal repo (有 .git/ 目录)

func (s *FileStore) discoverRepos() error {
    entries, err := os.ReadDir(s.cfg.DataPath)
    if err != nil {
        return err
    }

    for _, entry := range entries {
        if !entry.IsDir() {
            continue
        }

        name := entry.Name()
        fullPath := filepath.Join(s.cfg.DataPath, name)

        // 跳过特殊目录
        if name == ".lfs" || strings.HasPrefix(name, ".") {
            continue
        }

        // 检查是否是 Git 仓库
        isBare, isRepo := s.checkRepo(fullPath)
        if !isRepo {
            continue
        }

        repoName := strings.TrimSuffix(name, ".git")
        meta := s.loadRepoMeta(fullPath)

        s.repos[repoName] = &Repo{
            Name:        repoName,
            Path:        fullPath,
            IsBare:      isBare,
            Description: meta.Description,
            Private:     meta.Private,
        }
    }

    return nil
}

func (s *FileStore) Repositories(ctx context.Context) ([]*models.Repo, error) {
    s.mu.RLock()
    defer s.mu.RUnlock()

    repos := make([]*models.Repo, 0, len(s.repos))
    for _, repo := range s.repos {
        repos = append(repos, repo.ToModel())
    }
    return repos, nil
}
```

### 仓库元数据

```go
// 仓库元数据存储在仓库目录下的 .soft-serve.json

type RepoMeta struct {
    Description   string         `json:"description"`
    Private       bool           `json:"private"`
    ProjectName   string         `json:"project_name"`
    Collaborators []Collaborator `json:"collaborators"`
    Webhooks      []Webhook      `json:"webhooks"`
}

func (s *FileStore) loadRepoMeta(repoPath string) *RepoMeta {
    metaPath := filepath.Join(repoPath, ".soft-serve.json")
    data, err := os.ReadFile(metaPath)
    if err != nil {
        return &RepoMeta{}
    }

    var meta RepoMeta
    json.Unmarshal(data, &meta)
    return &meta
}

func (s *FileStore) saveRepoMeta(repoPath string, meta *RepoMeta) error {
    metaPath := filepath.Join(repoPath, ".soft-serve.json")
    data, err := json.MarshalIndent(meta, "", "  ")
    if err != nil {
        return err
    }
    return os.WriteFile(metaPath, data, 0644)
}
```

## 后端功能对比

| 功能 | FileStore | DBStore |
|------|-----------|---------|
| **用户管理** | 目录结构 | 数据库表 |
| **公钥管理** | 文件 | 数据库表 |
| **仓库发现** | 自动扫描 | 数据库 + 目录 |
| **仓库元数据** | `.soft-serve.json` | 数据库表 |
| **协作者** | 配置文件 | 数据库表 |
| **Webhook** | 配置文件 | 数据库表 |
| **访问令牌** | ❌ 不支持 | ✅ |
| **LFS** | ✅ | ✅ |
| **仓库镜像** | ❌ | ✅ |
| **系统设置** | ❌ (config.yaml) | ✅ |

## Store 接口适配

FileStore 需要实现所有 store 接口，但不支持的功能返回 `ErrNotSupported`:

```go
// pkg/store/file/token.go
package file

func (s *FileStore) CreateAccessToken(...) (*models.AccessToken, error) {
    return nil, store.ErrNotSupported
}

func (s *FileStore) AccessTokenByToken(...) (*models.AccessToken, error) {
    return nil, store.ErrNotSupported
}
```
