# 10 - 聊天功能 (Chat)

## 概述

基于现有 SSH 结构，添加类似 IRC 的聊天机制。用户通过 SSH 公钥认证后，使用 `/chat` 命令进入聊天模式。

## 需求确认

| 需求项 | 决定 |
|--------|------|
| 身份认证 | 使用现有 SSH 公钥认证 |
| 消息持久化 | 持久化到文件系统（JSONL 格式） |
| 频道持久化 | 持久化到文件系统（JSONL 格式，追加写入） |
| 历史查询 | 默认 7 天，支持时间范围参数 |
| 频道管理 | 混合模式（默认 #general + 用户自建） |
| 界面形式 | 简单命令行 |
| 高级特性 | 仅 /topic 频道主题 |

## 设计约束（简化实现）

| 约束项 | 说明 |
|--------|------|
| 消息已读状态 | 不提供 |
| 消息 ID | 时间戳 + 流水号 |
| 频道 topic 权限 | 任何人都可设置 |
| 频道/消息删除 | 不支持，一旦写入不可删除 |
| 用户在线状态 | 不监控 |
| 消息撤回/编辑 | 不支持 |
| 事件与消息混合 | 刻意设计，便于回放完整历史 |

## 核心命令

```
# 进入聊天模式
> chat

# 频道管理
/join #channel        # 加入/创建频道
/leave #channel       # 离开频道
/channels             # 列出所有频道

# 切换当前目标
/select #general      # 切换到频道
/select @bob          # 切换到与 bob 的私聊

# 发送消息
hello                 # 发送到当前目标（频道或私聊）
#general hello        # 直接发送到 #general（不切换当前目标）

# 私聊（不切换当前目标）
/dm @bob hello        # 发送私聊给 bob

# 历史消息
/history              # 查看当前目标最近 7 天历史
/history #channel     # 查看指定频道历史
/history @bob         # 查看与 bob 的私聊历史
/history --from 2025-01-01 --to 2025-01-15  # 指定时间范围

# 频道信息
/topic #channel       # 查看频道主题
/topic #channel xxx   # 设置频道主题（任何人可设置）

# 其他
/who                  # 列出当前频道用户
/names #channel       # 列出指定频道用户
/whois @name          # 查看用户详细信息
/exit 或 Ctrl+D       # 退出聊天模式
```

```go
// pkg/chat/channel.go

type ChannelEventType string

const (
    ChannelEventJoin    ChannelEventType = "join"
    ChannelEventLeave   ChannelEventType = "leave"
    ChannelEventTopic   ChannelEventType = "topic"
    ChannelEventMembers ChannelEventType = "members"  // 成员快照
)

// ChannelEvent 频道事件（JSONL 记录）
type ChannelEvent struct {
    Type        ChannelEventType `json:"type"`         // 事件类型
    Timestamp   time.Time        `json:"timestamp"`
    User        string           `json:"user,omitempty"`    // 触发用户
    Topic       string           `json:"topic,omitempty"`   // 主题变更时
    Members     []string         `json:"members,omitempty"` // 成员快照
    Date        string           `json:"date,omitempty"`    // 日期快照 (YYYY-MM-DD)
}

// Channel 频道元数据（内存中维护）
type Channel struct {
    Name        string    `json:"name"`         // #general
    Topic       string    `json:"topic"`        // 频道主题
    Creator     string    `json:"creator"`      // 创建者用户名
    CreatedAt   time.Time `json:"created_at"`
    Members     []string  `json:"members"`      // 当前成员列表（动态计算）
}

// Members 快照规则：
// 1. 每次 /join 或 /leave 后，追加一条 ChannelEvent 记录当前 members
// 2. 每天第一个消息发出后，追加一条 members 快照（无消息则不更新）
// 3. 读取时从后向前扫描，取最近一次 members 快照

// 系统默认频道
var DefaultChannels = []string{"#general"}
```

```go
type MessageType int

const (
    MsgTypeChannel MessageType = iota  // 频道消息
    MsgTypePrivate                      // 私聊消息
    MsgTypeSystem                       // 系统消息（加入/离开通知等）
)

type Message struct {
    ID        string      `json:"id"`           // 时间戳+流水号，如 "20250115100200-001"
    Type      MessageType `json:"type"`
    From      string      `json:"from"`         // 发送者
    To        string      `json:"to"`           // 接收目标：#channel 或 @user
    Content   string      `json:"content"`
    Timestamp time.Time   `json:"timestamp"`
}

// ID 格式：YYYYMMDDHHmmss-NNN（精确到秒 + 3位流水号）
// 示例：20250115100200-001, 20250115100200-002
// 同一秒内最多支持 999 条消息

// To 字段说明：
// - 频道消息：To = "#general", "#random" 等
// - 私聊消息：To = "@bob"（接收方）
// - 统一使用 To 字段，通过前缀区分类型
```

```go
type TargetType int

const (
    TargetTypeChannel TargetType = iota  // 频道
    TargetTypePrivate                    // 私聊
)

type ChatUser struct {
    Name         string      // 用户名（来自 SSH 认证）
    PublicKey    string      // 公钥指纹
    CurrentType  TargetType  // 当前目标类型
    CurrentCh    string      // 当前频道（CurrentType == Channel）
    CurrentPeer  string      // 当前私聊对象（CurrentType == Private）
    JoinedChs    []string    // 已加入的频道列表
    conn         *ssh.ServerConn
}

// CurrentTarget 返回当前目标的字符串表示
func (u *ChatUser) CurrentTarget() string {
    if u.CurrentType == TargetTypeChannel {
        return u.CurrentCh
    }
    return "@" + u.CurrentPeer
}
```

```go
// pkg/chat/store.go

type ChatStore interface {
    // 频道操作
    GetChannel(name string) (*Channel, error)
    ListChannels() ([]*Channel, error)

    // 频道事件操作（追加到 #channel.jsonl）
    AppendChannelEvent(channel string, event *ChannelEvent) error
    GetChannelMembers(channel string) ([]string, error)  // 从最近快照获取成员
    GetLastMessageDate(channel string) (time.Time, error) // 获取最后一条消息的日期

    // 消息操作
    AppendChannelMessage(channel string, msg *Message) error   // 写入 #channel.jsonl
    AppendPrivateMessage(user1, user2 string, msg *Message) error  // 写入 dm/user1-user2.jsonl
    GetChannelMessages(channel string, opts GetMessagesOptions) ([]*Message, error)
    GetPrivateMessages(user1, user2 string, opts GetMessagesOptions) ([]*Message, error)

    // 生命周期
    Init() error     // 创建默认频道等
    Shutdown() error // 优雅关闭，记录最终快照
}

// GetMessagesOptions 消息查询选项
type GetMessagesOptions struct {
    // 默认查询最近 7 天
    // 需要更久历史时，传入 StartTime 和 EndTime
    StartTime *time.Time  // 开始时间（可选）
    EndTime   *time.Time  // 结束时间（可选）
    Limit     int         // 限制数量（可选，默认 100）
}
```

```go
// Members 快照逻辑：发送消息前检查
func (s *FileStore) appendChannelMessageWithSnapshot(ch string, msg *Message) error {
    lastDate, _ := s.GetLastMessageDate(ch)
    today := time.Now().Format("2006-01-02")

    if lastDate != today {
        // 跨日，先补充 members 快照
        members := s.GetCurrentMembers(ch)
        s.AppendChannelEvent(ch, &ChannelEvent{
            Type:      ChannelEventMembers,
            Timestamp: time.Now(),
            Members:   members,
            Date:      today,
        })
    }

    return s.AppendChannelMessage(ch, msg)
}
```

```go
// pkg/chat/commands.go

type InputType int

const (
    InputTypeCommand InputType = iota  // /xxx 命令
    InputTypeChannel                   // #channel xxx 频道消息
    InputTypeMessage                   // xxx 当前目标消息
)

type ParsedInput struct {
    Type    InputType
    Target  string   // 目标：#channel（仅频道消息类型）
    Content string   // 消息内容
    Command string   // 命令名（仅命令类型）
    Args    []string // 命令参数
}

func ParseInput(input string) *ParsedInput {
    input = strings.TrimSpace(input)

    // 命令：/xxx
    if strings.HasPrefix(input, "/") {
        parts := strings.Fields(input[1:])
        return &ParsedInput{
            Type:    InputTypeCommand,
            Command: parts[0],
            Args:    parts[1:],
        }
    }

    // 频道消息：#channel xxx
    if strings.HasPrefix(input, "#") {
        parts := strings.SplitN(input[1:], " ", 2)
        if len(parts) == 2 {
            return &ParsedInput{
                Type:    InputTypeChannel,
                Target:  "#" + parts[0],
                Content: parts[1],
            }
        }
    }

    // 当前目标消息
    return &ParsedInput{
        Type:    InputTypeMessage,
        Content: input,
    }
}

// 命令处理映射
var commands = map[string]CommandHandler{
    "join":     handleJoin,

    "channels": handleChannels,
    "who":      handleWho,
    "names":    handleNames,
    "whois":    handleWhois,
    "exit":     handleExit,
    "quit":     handleExit,
}
```

## 风险评估

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
| 并发消息写入 | 中 | 使用 sync.Mutex 保护文件写入 |
| 大量历史消息 | 低 | 默认查询 7 天，支持时间范围参数 |
| 断线重连 | 低 | 会话状态可丢弃，用户重新加入 |
| 存储格式兼容 | 中 | 使用 JSONL，版本化处理 |
| Members 快照一致 | 中 | join/leave 必记录；首条消息检查跨日补充 |
| 优雅关闭失败 | 中 | 监听信号，确保最终快照写入 |

## 优雅关闭

```go
// pkg/chat/chat.go

func (c *Chat) Shutdown(ctx context.Context) error {
    // 1. 停止接受新连接
    c.mu.Lock()
    c.shuttingDown = true
    c.mu.Unlock()

    // 2. 通知所有在线用户
    for _, sess := range c.sessions {
        sess.term.Write([]byte("\nServer is shutting down. Goodbye!\n"))
        close(sess.done)
    }

    // 3. 等待所有会话结束或超时
    done := make(chan struct{})
    go func() {
        for len(c.sessions) > 0 {
            time.Sleep(100 * time.Millisecond)
        }
        close(done)
    }()

    select {
    case <-done:
    case <-ctx.Done():
        // 超时，强制关闭
    }

    // 4. 为所有活跃频道记录最终 members 快照
    for name, ch := range c.channels {
        c.store.AppendChannelEvent(name, &ChannelEvent{
            Type:      ChannelEventMembers,
            Timestamp: time.Now(),
            Members:   ch.Members,
            Date:      time.Now().Format("2006-01-02"),
        })
    }

    return c.store.Shutdown()
}
```
