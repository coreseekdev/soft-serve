# 04 - 用户管理

## 需求回顾

1. 配置包含 `user_home`，每个子目录都是有效的用户
2. 用户通过 SSH 登录需要 `.ssh/*.pub` 公钥文件
3. 一个用户可以有多个公钥身份
4. Admin 用户默认使用 `~/.ssh/id_*.pub` 认证，可通过配置重新指定

## 目录结构

```
~/.soft-serve-git/users/              # 用户目录 (可配置)
│
├── alice/                            # 用户 alice
│   ├── .ssh/

└── deploy/                           # 用户 deploy (CI/CD 用)
    └── .ssh/
        └── deploy_key.pub           # 公钥

~/.ssh/                               # Admin 公钥位置 (可配置)
├── id_ed25519.pub                    # 默认 admin 认证
└── id_rsa.pub                        # 备用 admin 公钥
```

## 代码实现

### 用户管理器

```go
// pkg/store/file/user.go
package file

import (
    "context"
    "os"
    "path/filepath"
    "strings"
    "time"

    "github.com/charmbracelet/soft-serve/pkg/sshutils"
    "golang.org/x/crypto/ssh"
)

type UserManager struct {
    usersDir   string         // 用户目录
    adminDir   string         // Admin 公钥目录
    users      map[string]*User
    adminKeys  []ssh.PublicKey
    mu         sync.RWMutex
}

func NewUserManager(usersDir, adminDir string) (*UserManager, error) {
    um := &UserManager{
        usersDir:  usersDir,
        adminDir:  adminDir,
        users:     make(map[string]*User),
    }

    // 加载 admin 公钥
    if err := um.loadAdminKeys(); err != nil {
        return nil, err
    }

    // 加载用户
    if err := um.loadUsers(); err != nil {
        return nil, err
    }

    return um, nil
}

// loadAdminKeys 加载 admin 公钥
func (um *UserManager) loadAdminKeys() error {
    entries, err := os.ReadDir(um.adminDir)
    if os.IsNotExist(err) {
        return nil
    }
    if err != nil {
        return err
    }

    for _, entry := range entries {
        if entry.IsDir() {
            continue
        }

        name := entry.Name()
        // 只读取 id_*.pub 文件
        if !strings.HasPrefix(name, "id_") || !strings.HasSuffix(name, ".pub") {
            continue
        }

        data, err := os.ReadFile(filepath.Join(um.adminDir, name))
        if err != nil {
            continue
        }

        key, _, _, _, err := ssh.ParseAuthorizedKey(data)
        if err != nil {
            continue
        }

        um.adminKeys = append(um.adminKeys, key)
    }

    return nil
}

// loadUsers 加载所有用户
func (um *UserManager) loadUsers() error {
    entries, err := os.ReadDir(um.usersDir)
    if os.IsNotExist(err) {
        // 创建目录
        return os.MkdirAll(um.usersDir, 0755)
    }
    if err != nil {
        return err
    }

    for _, entry := range entries {
        if !entry.IsDir() {
            continue
        }

        username := entry.Name()
        userDir := filepath.Join(um.usersDir, username)
        sshDir := filepath.Join(userDir, ".ssh")

        // 读取公钥
        keys, err := um.loadUserKeys(sshDir)
        if err != nil || len(keys) == 0 {
            continue // 没有公钥的用户无效
        }

        // 加载可选的元数据
        user := &User{
            Username:  username,
            Keys:      keys,
            IsAdmin:   false,
            CreatedAt: time.Now(),
        }

        um.loadUserMeta(user, userDir)
        um.users[username] = user
    }

    return nil
}

// loadUserKeys 加载用户公钥
func (um *UserManager) loadUserKeys(sshDir string) ([]string, error) {
    entries, err := os.ReadDir(sshDir)
    if err != nil {
        return nil, err
    }

    var keys []string
    for _, entry := range entries {
        if entry.IsDir() {
            continue
        }

        name := entry.Name()
        if !strings.HasSuffix(name, ".pub") {
            continue
        }

        data, err := os.ReadFile(filepath.Join(sshDir, name))
        if err != nil {
            continue
        }

        key := strings.TrimSpace(string(data))
        // 验证公钥格式
        if _, _, err := sshutils.ParseAuthorizedKey(key); err == nil {
            keys = append(keys, key)
        }
    }

    return keys, nil
}

// loadUserMeta 加载用户元数据
func (um *UserManager) loadUserMeta(user *User, userDir string) {
    metaPath := filepath.Join(userDir, ".soft-serve-user.json")
    data, err := os.ReadFile(metaPath)
    if err != nil {
        return
    }

    var meta struct {
        DisplayName string    `json:"display_name"`
        Email       string    `json:"email"`
        Admin       bool      `json:"admin"`
        CreatedAt   time.Time `json:"created_at"`
    }

    if err := json.Unmarshal(data, &meta); err != nil {
        return
    }

    user.DisplayName = meta.DisplayName
    user.Email = meta.Email
    user.IsAdmin = meta.Admin
    if !meta.CreatedAt.IsZero() {
        user.CreatedAt = meta.CreatedAt
    }
}

// Authenticate 认证用户
func (um *UserManager) Authenticate(pubKey ssh.PublicKey) (*User, bool) {
    um.mu.RLock()
    defer um.mu.RUnlock()

    // 1. 检查是否是 admin
    for _, adminKey := range um.adminKeys {
        if ssh.KeysEqual(pubKey, adminKey) {
            return &User{
                Username: "admin",
                IsAdmin:  true,
            }, true
        }
    }

    // 2. 检查普通用户
    for _, user := range um.users {
        for _, keyStr := range user.Keys {
            userKey, _, _, _, err := ssh.ParseAuthorizedKey([]byte(keyStr))
            if err != nil {
                continue
            }
            if ssh.KeysEqual(pubKey, userKey) {
                return user, true
            }
        }
    }

    // 3. 匿名用户
    return nil, false
}

// CreateUser 创建用户
func (um *UserManager) CreateUser(username string) (*User, error) {
    um.mu.Lock()
    defer um.mu.Unlock()

    if _, exists := um.users[username]; exists {
        return nil, ErrUserExists
    }

    userDir := filepath.Join(um.usersDir, username)
    sshDir := filepath.Join(userDir, ".ssh")

    if err := os.MkdirAll(sshDir, 0700); err != nil {
        return nil, err
    }

    user := &User{
        Username:  username,
        CreatedAt: time.Now(),
        Keys:      []string{},
    }

    um.users[username] = user
    return user, nil
}

// AddUserKey 添加用户公钥
func (um *UserManager) AddUserKey(username, key string) error {
    um.mu.Lock()
    defer um.mu.Unlock()

    user, exists := um.users[username]
    if !exists {
        return ErrUserNotFound
    }

    // 验证公钥
    if _, _, err := sshutils.ParseAuthorizedKey(key); err != nil {
        return ErrInvalidKey
    }

    // 写入文件
    keyHash := keyHash(key) // 生成唯一文件名
    keyPath := filepath.Join(um.usersDir, username, ".ssh", keyHash+".pub")
    if err := os.WriteFile(keyPath, []byte(key+"\n"), 0644); err != nil {
        return err
    }

    user.Keys = append(user.Keys, key)
    return nil
}

// ListUsers 列出所有用户
func (um *UserManager) ListUsers() []*User {
    um.mu.RLock()
    defer um.mu.RUnlock()

    users := make([]*User, 0, len(um.users))
    for _, user := range um.users {
        users = append(users, user)
    }
    return users
}
```

## SSH 命令

### 用户管理命令

```bash
# 创建用户
ssh localhost user create alice

# 添加公钥
ssh localhost user add-key alice "ssh-ed25519 AAAA..."

# 列出用户
ssh localhost user list

# 查看用户信息
ssh localhost user info alice

# 删除用户
ssh localhost user delete alice
```

## 集成到 Soft Serve

```go
// pkg/ssh/middleware.go

func (mw *Middleware) Authenticate(conn ssh.ConnMetadata, pubKey ssh.PublicKey) (*ssh.Permissions, error) {
    // 使用 UserManager 认证
    user, ok := mw.userManager.Authenticate(pubKey)
    if !ok {
        // 允许匿名访问
        return &ssh.Permissions{
            Extensions: map[string]string{
                "access": "read-only",
            },
        }, nil
    }

    access := "read-write"
    if user.IsAdmin {
        access = "admin"
    }

    return &ssh.Permissions{
        Extensions: map[string]string{
            "user":    user.Username,
            "access":  access,
        },
    }, nil
}
```

## 安全考虑

1. **公钥验证**: 所有公钥必须通过 `ssh.ParseAuthorizedKey` 验证
2. **目录权限**: 用户 `.ssh/` 目录权限应为 700，公钥文件为 644
3. **Admin 保护**: Admin 公钥目录不应与用户目录相同
4. **热重载**: 支持运行时重新加载用户（可选）
