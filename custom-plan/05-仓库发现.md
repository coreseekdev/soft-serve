# 05 - 仓库发现

## 需求回顾

1. Soft Serve 默认为当前目录下所有 Git repo 的子目录服务
2. 支持 bare repo (`xxx.git/`) 和普通 repo (`xxx/`)
3. LFS 对象存储在 serve 目录下的 `.lfs/`，可配置

## 仓库类型

| 类型 | 目录名 | 特征 | 说明 |
|------|--------|------|------|
| Bare Repo | `xxx.git/` | 有 `HEAD` 文件，无 `.git/` 目录 | 服务端标准格式 |
| Normal Repo | `xxx/` | 有 `.git/` 目录 | 可同时承载日常开发 |

## 代码实现

### 仓库发现器

```go
// pkg/discovery/discovery.go
package discovery

import (
    "os"
    "path/filepath"
    "strings"
    "time"
)

type Repo struct {
    Name        string
    Path        string
    IsBare      bool
    Description string
    Private     bool
    ProjectName string
    UpdatedAt   time.Time
}

type Discoverer struct {
    serveDir string
    repos    map[string]*Repo
    lfsDir   string
}

func NewDiscoverer(serveDir, lfsDir string) *Discoverer {
    return &Discoverer{
        serveDir: serveDir,
        lfsDir:   lfsDir,
        repos:    make(map[string]*Repo),
    }
}

// Discover 发现并注册所有仓库
func (d *Discoverer) Discover() error {
    entries, err := os.ReadDir(d.serveDir)
    if err != nil {
        return err
    }

    for _, entry := range entries {
        if !entry.IsDir() {
            continue
        }

        name := entry.Name()
        fullPath := filepath.Join(d.serveDir, name)

        // 跳过 LFS 目录
        if name == ".lfs" {
            continue
        }

        // 跳过隐藏目录
        if strings.HasPrefix(name, ".") {
            continue
        }

        // 检查是否是仓库
        isBare, isRepo := d.checkRepo(fullPath)
        if !isRepo {
            continue
        }

        // 规范化仓库名
        repoName := strings.TrimSuffix(name, ".git")

        // 加载元数据
        meta := d.loadMeta(fullPath)

        // 获取修改时间
        info, _ := entry.Info()
        updatedAt := time.Now()
        if info != nil {
            updatedAt = info.ModTime()
        }

        d.repos[repoName] = &Repo{
            Name:        repoName,
            Path:        fullPath,
            IsBare:      isBare,
            Description: meta.Description,
            Private:     meta.Private,
            ProjectName: meta.ProjectName,
            UpdatedAt:   updatedAt,
        }
    }

    return nil
}

// checkRepo 检查目录是否是 Git 仓库
// 返回: (是否 bare, 是否是仓库)
func (d *Discoverer) checkRepo(path string) (bool, bool) {
    // 检查 bare repo: 存在 HEAD 文件且没有 .git 目录
    headPath := filepath.Join(path, "HEAD")
    gitPath := filepath.Join(path, ".git")

    hasHead := fileExists(headPath)
    hasGitDir := dirExists(gitPath)

    if hasHead && !hasGitDir {
        // Bare repo
        return true, true
    }

    if hasGitDir {
        // Normal repo
        return false, true
    }

    return false, false
}

// loadMeta 加载仓库元数据
func (d *Discoverer) loadMeta(repoPath string) *RepoMeta {
    metaPath := filepath.Join(repoPath, ".soft-serve.json")

    data, err := os.ReadFile(metaPath)
    if err != nil {
        return &RepoMeta{}
    }

    var meta RepoMeta
    if err := json.Unmarshal(data, &meta); err != nil {
        return &RepoMeta{}
    }

    return &meta
}

// GetRepo 获取仓库
func (d *Discoverer) GetRepo(name string) (*Repo, bool) {
    repo, ok := d.repos[name]
    return repo, ok
}

// ListRepos 列出所有仓库
func (d *Discoverer) ListRepos() []*Repo {
    repos := make([]*Repo, 0, len(d.repos))
    for _, repo := range d.repos {
        repos = append(repos, repo)
    }
    return repos
}

// Refresh 刷新仓库列表
func (d *Discoverer) Refresh() error {
    d.repos = make(map[string]*Repo)
    return d.Discover()
}

func fileExists(path string) bool {
    _, err := os.Stat(path)
    return err == nil
}

func dirExists(path string) bool {
    info, err := os.Stat(path)
    return err == nil && info.IsDir()
}
```

```go
// pkg/discovery/ignore.go
package discovery

import (
    "bufio"
    "os"
    "path/filepath"
    "strings"

    "github.com/gobwas/glob"
)

type IgnoreMatcher struct {
    patterns []glob.Glob
}

func LoadIgnore(serveDir string) (*IgnoreMatcher, error) {
    ignorePath := filepath.Join(serveDir, ".soft-serve-ignore")

    file, err := os.Open(ignorePath)
    if err != nil {
        return &IgnoreMatcher{}, nil // 文件不存在，无忽略规则
    }
    defer file.Close()

    var patterns []glob.Glob
    scanner := bufio.NewScanner(file)
    for scanner.Scan() {
        line := strings.TrimSpace(scanner.Text())

        // 跳过空行和注释
        if line == "" || strings.HasPrefix(line, "#") {
            continue
        }

        // 解析 glob 模式
        if g, err := glob.Compile(line); err == nil {
            patterns = append(patterns, g)
        }
    }

    return &IgnoreMatcher{patterns: patterns}, nil
}

func (m *IgnoreMatcher) ShouldIgnore(name string) bool {
    for _, pattern := range m.patterns {
        if pattern.Match(name) {
            return true
        }
    }
    return false
}
```

## LFS 存储

```go
// pkg/lfs/storage.go
package lfs

import (
    "os"
    "path/filepath"
)

type LFSStorage struct {
    basePath string
}

func NewLFSStorage(basePath string) *LFSStorage {
    return &LFSStorage{basePath: basePath}
}

// ObjectPath 返回 LFS 对象路径
// 结构: .lfs/objects/{oid[:2]}/{oid}/data
func (s *LFSStorage) ObjectPath(oid string) string {
    if len(oid) < 2 {
        oid = "00" + oid
    }
    return filepath.Join(s.basePath, "objects", oid[:2], oid, "data")
}

// Exists 检查对象是否存在
func (s *LFSStorage) Exists(oid string) bool {
    _, err := os.Stat(s.ObjectPath(oid))
    return err == nil
}

// Write 写入对象
func (s *LFSStorage) Write(oid string, data []byte) error {
    path := s.ObjectPath(oid)

    // 创建目录
    if err := os.MkdirAll(filepath.Dir(path), 0755); err != nil {
        return err
    }

    // 写入临时文件
    tmpPath := path + ".tmp"
    if err := os.WriteFile(tmpPath, data, 0644); err != nil {
        return err
    }

    // 原子重命名
    return os.Rename(tmpPath, path)
}

// Read 读取对象
func (s *LFSStorage) Read(oid string) ([]byte, error) {
    return os.ReadFile(s.ObjectPath(oid))
}

// Size 返回对象大小
func (s *LFSStorage) Size(oid string) (int64, error) {
    info, err := os.Stat(s.ObjectPath(oid))
    if err != nil {
        return 0, err
    }
    return info.Size(), nil
}
```

## 使用示例

### 启动服务

```bash
# 在当前目录启动服务
cd ~/projects
soft serve

# 服务以下仓库:
# - ~/projects/project-a.git (bare)
# - ~/projects/project-b (normal)
# - ~/projects/tools.git (bare)
```

### 指定服务目录

```bash
# 通过环境变量
SOFT_SERVE_SERVE_DIR=/var/git soft serve

# 或在配置中指定
SOFT_SERVE_CONFIG='{"serve_dir":"/var/git"}' soft serve
```

### 克隆仓库

```bash
# 克隆 bare repo
git clone ssh://localhost:23231/project-a.git

# 克隆 normal repo
git clone ssh://localhost:23231/project-b.git
```

### 添加仓库

```bash
# 创建 bare repo
mkdir my-repo.git
cd my-repo.git
git init --bare

# 或直接推送创建
git remote add origin ssh://localhost:23231/new-repo.git
git push -u origin main
```
